// 버튼 위에 마우스를 올리면 원의 색상이 변경됨

let c, button;

function setup() {
  createCanvas(500, 500);
  button = createButton("Click!"); 		// 버튼 생성
  button.position(10, 10); 			// 버튼 위치 설정
  button.size(150, 50); 			// 버튼 크기 설정
  button.style('font-size', '20px');		// 버튼 폰트 크기 설정
  button.style('color', '#ff0000');			// 버튼 폰트 컬러 설정
  button.style('background-color', '#ffff00');	// 버튼 배경 컬러 설정
  button.style('border', '5px solid');		// 버튼 테두리 두께와 모양 설정
  button.style('border-color', 'rgb(0,0,0)');		// 버튼 테두리 색상 설정
  button.style('border-radius', '50%');		// 버튼 테두리 곡선 설정
  button.mouseOver(colorChange); 		// 버튼 위에 올리면 colorChange() 호출
  c = color(100);
}

function draw() {
  background(200);
  fill(c);
  ellipse(width/2, height/2, 300, 300);
}

function colorChange() {			// 콜백함수: colorChange()
  c = color(random(255), random(255), random(255));
  button.style('color', c);			// 버튼 폰트 컬러 변경
  button.html('Click again!');			// 버튼 텍스트 변경
}
