// 좌우 반사하는 원

let x = 0, velocity = 1;

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  bounce();
  display();	
  move();
}

function bounce() {
  if ( (x < 0) || (x > width) )
    velocity *= -1;
}

function display() {
  ellipse(x, height/2, 30, 30);
  print(x);
}

function move() {
  x += velocity;
}
