// 원의 마찰 이동 (p5.Vector 클래스 미사용)

let x = 0, y = 0;
let velocity_x = 10, velocity_y = 10;
let friction = 0.98;

function setup() {
  createCanvas(500, 500);
  background(0);
}

function draw() {
  ellipse(x, y, 30, 30);
  velocity_x *= friction;
  velocity_y *= friction;
  x += velocity_x;
  y += velocity_y;
}
