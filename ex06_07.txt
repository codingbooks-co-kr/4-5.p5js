// 마우스의 좌우 버튼을 누르거나 떼면 원의 색상 변경 (color() 함수 사용)

let c;

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  if (mouseIsPressed && (mouseButton === LEFT)) {
    c = color(255, 0, 0);
  } else if (mouseIsPressed && (mouseButton === RIGHT)) {
    c = color(0, 255, 0);
  } else {
    c = color(0, 0, 255);
  }
  fill(c);
  ellipse(width/2, height/2, 200, 200);
}
