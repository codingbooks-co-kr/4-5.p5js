// 중력과 마찰력을 받는 원이 낙하하고 바닥에서 반사하면서 마찰력으로 점점 감쇄

let position, velocity, accel, gravity;
let d = 30, friction = 0.99;

function setup() {
  createCanvas(500, 500);
  position = createVector(width/2, 50);
  velocity = createVector(0, 0);
  accel = createVector(0, 0);
  gravity = createVector(0.0, 0.1);
  firstForce();
}

function draw() {
  // 페이딩 효과
  noStroke();
  fill(0, 10);
  rect(0, 0, width, height);

  fill(255);
  ellipse(position.x, position.y, d, d);

  accel.add(gravity);
  velocity.mult(friction);
  velocity.add(accel);
  position.add(velocity);
  accel.mult(0);

  // 경계에 부딪힐 때 반사 적용
  if (position.x < 0) {
    position.x = 0;
    velocity.x *= -1;
  }
  if (position.x > width) {
    position.x = width;
    velocity.x *= -1;
  }
  if (position.y < 0) {
    position.y = 0;
    velocity.y *= -1;
  }
  if (position.y > height) {
    position.y = height;
    velocity.y *= -1;
  }
}

function firstForce() {
  let force = createVector(random(-5, 5), random(-5, 5));
  accel.add(force);
}
