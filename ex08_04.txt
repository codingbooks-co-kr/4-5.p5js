// 랜덤 생성되는 원으로 큰 원 문양 만들기, 마우스 위치에 따라 생성되는 원의 개수 조정

function setup() {
  createCanvas(300, 300);
}

function draw() {  
  background(0);
  let num = map(mouseX, 0, width, 200, 2000);
  for (let i = 0; i < num; i++) {
    let x = random(width);
    let y = random(height);
    let d = dist(width/2, height/2, x, y);
    if (d < width/2) {
      fill(255, 255, 0);
    } else {
      fill(255, 0, 0);
    }
    ellipse(x, y, 10, 10);
  }
}
