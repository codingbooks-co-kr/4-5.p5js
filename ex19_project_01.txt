// 입력된 이름은 랜덤한 색상과 속도로 상하좌우 반사. 다섯 개를 넘으면 0번 요소 삭제

let list = [];

function setup() {
  createCanvas(500, 500);
}

function draw() {
  background(0);
  for (let i = 0; i < list.length; i++) {
    let n = list[i];
    n.display();
    n.move();
    n.bounce();
  }

  if (list.length > 5) {
    list.splice(0, 1);
  }
}

function mousePressed() {
  let n = new Name(mouseX, mouseY);
  n.name = prompt("Enter your name");
  list.push(n);
}

// Name 클래스 코드

class Name {
  constructor(x, y) {
    this.position = createVector(x, y);
    this.velocity = createVector(random(-5, 5), random(-5, 5));
    this.col = color(random(255), random(255), random(255));
    this.name = "";
    this.d = 50;
  }

  display() {
    fill(this.col);
    textSize(this.d);
    textAlign(CENTER);
    text(this.name, this.position.x, this.position.y);
  }

  move() {
    this.position.add(this.velocity);
  }

  bounce() {
    // 경계에 부딪힐 때 반사 적용
    if (this.position.x < 0) {
      this.position.x = 0;
      this.velocity.x *= -1;
    }
    if (this.position.x > width) {
      this.position.x = width;
      this.velocity.x *= -1;
    }
    if (this.position.y < 0) {
      this.position.y = 0;
      this.velocity.y *= -1;
    }
    if (this.position.y > height) {
      this.position.y = height;
      this.velocity.y *= -1;
    }
  }
}
