// 본문 코드: 원과 사각형 좌우 반사
// 상속 적용 (완성)

let c, s;

function setup() {
  createCanvas(300, 300);
  c = new Circle(255, width/2, 100, 30, -2);
  s = new Square(255, width/2, 200, 30, 2);
}

function draw() {
  background(0);
  c.display();
  c.move();
  s.display();
  s.move();
}

// 클래스 코드

class Shape {
  constructor(c, x, y, d, velocity) {
    this.c = c;
    this.x = x;
    this.y = y;
    this.d = d;
    this.velocity = velocity;
  }

  move() {
    this.x += this.velocity;
    if (this.x < 0 || this.x > width) {
      this.velocity *= -1;
    }
  }
}

class Circle extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(this.c, 0, 0);
    ellipse(this.x, this.y, this.d, this.d);
  }
}

class Square extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(0, this.c, 0);
    rect(this.x, this.y, this.d, this.d);
  }
}
