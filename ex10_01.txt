// 마우스의 좌우 위치에 따라 이동 속도가 가감되는 10개의 원 

let num = 10;
let x = [], y = [];

function setup() {
  createCanvas(500, 300);
  for (let i = 0; i < num; i++) {
    x[i] = random(-500, 0);
    y[i] = random(height);
  }
}

function draw() {
  background(0);
  let velocity = map(mouseX, 0, width, 0.5, 5);
  for (let i = 0; i < num; i++) {
    ellipse(x[i], y[i], 30, 30);
    x[i] += velocity;
  }
}
