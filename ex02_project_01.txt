// 선, 원, 사각형 및 문자열을 사용하여 자유롭게 표현

function setup() {
  createCanvas(400, 400);
  background(0);
  
  stroke(255, 0, 0); 
  line(0, 0, 400, 400);
  line(400, 0, 0, 400);

  strokeWeight(10);
  stroke(0, 255, 0); 
  noFill();
  ellipse(200, 200, 150, 150);

  strokeWeight(20);
  stroke(0, 0, 255); 
  noFill();
  rectMode(CENTER);
  rect(200, 200, 300, 300);

  strokeWeight(1);
  fill(255, 255, 0);
  textSize(30);
  textAlign(CENTER);
  text('Hello, p5.js!', 200, 200);
}
