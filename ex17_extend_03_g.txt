// “객체 리터럴” 기반 객체 생성
// 원 객체 좌우 반사

let circle;

function setup() {
  createCanvas(300, 300);
  circle = {				// 객체 리터럴
    c: 255,				// 속성은 키:값으로 구성
    x: width / 2,
    y: height / 2,
    d: 30,
    velocity: 1,

    display: function () {		// 메소드
      fill(this.c, 255, 0);
      ellipse(this.x, this.y, this.d, this.d);
    },

    move: function () {		// 메소드
      this.x += this.velocity;
      if (this.x < 0 || this.x > width) {
        this.velocity *= -1;
      }
    }
  };
}

function draw() {
  background(0);
  circle.display(); 			// 메소드 호출
  circle.move(); 			// 메소드 호출
}
