// 원의 가속 이동 (p5.Vector 클래스 사용)

let position;
let velocity;
let accel;

function setup() {
  createCanvas(500, 500);
  background(0);
  position = createVector(0, 0);
  velocity = createVector(0, 0);
  accel = createVector(0.2, 0.2);
}

function draw() {
  ellipse(position.x, position.y, 30, 30);
  velocity.add(accel);
  position.add(velocity);
}
