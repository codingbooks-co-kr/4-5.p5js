// 상하좌우로 분산하는 100개의 원

let num = 100;
let x = [], y = [], velocity_x = [], velocity_y = [];

function setup() {
  createCanvas(500, 500);
  for (let i = 0; i < num; i++) {
    x[i] = width / 2;
    y[i] = height / 2;
    velocity_x[i] = random(-5, 5);
    velocity_y[i] = random(-5, 5);
  }
}

function draw() {
  background(0);
  for (let i = 0; i < num; i++) {
    ellipse(x[i], y[i], 30, 30);
    x[i] += velocity_x[i];
    y[i] += velocity_y[i];
  }
}
