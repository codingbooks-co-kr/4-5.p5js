// 본문 코드: 원 객체 좌우 반사
// 객체 두 개 생성 (완성)

let c1, c2;

function setup() {
  createCanvas(300, 300);
  c1 = new Circle(0, 100);
  c2 = new Circle(50, 200);
}

function draw() {
  background(0);
  c1.display();
  c1.move();
  c2.display();
  c2.move();
}

// 클래스 코드

class Circle {
  constructor(x, y) {
    this.c = 255;
    this.x = x;
    this.y = y;
    this.d = 30;
    this.velocity = 1;
  }

  display() {
    fill(this.c, 255, 0);
    ellipse(this.x, this.y, this.d, this.d);
  }

  move() {
    this.x += this.velocity;
    if (this.x < 0 || this.x > width) {
      this.velocity *= -1;
    }
  }
}
