// 가속력과 마찰력을 받으며 이동하는 원이 사각형 안에 들어가면 또 다른 마찰력 받기

let r;
let position, velocity, accel;
let d = 30, friction1 = 0.99, friction2 = 0.95;

function setup() {
  createCanvas(500, 500);
  r = new Rectangle(100, 200, 300, 100);
  position = createVector(width/2, 50);
  velocity = createVector(0, 0);
  accel = createVector(0, 0);
  firstForce();
}

function draw() {
  // 페이딩 효과
  noStroke();
  fill(0, 10);
  rect(0, 0, width, height);

  fill(255);
  r.display();

  fill(255, 0, 0);
  ellipse(position.x, position.y, d, d);

  if (r.contains(position.x, position.y)) {
    velocity.mult(friction2);
  } else {
    velocity.mult(friction1);
  }
  velocity.add(accel);
  position.add(velocity);
  accel.mult(0);

  // 경계에 부딪힐 때 반사 적용
  if (position.x < 0) {
    position.x = 0;
    velocity.x *= -1;
  }
  if (position.x > width) {
    position.x = width;
    velocity.x *= -1;
  }
  if (position.y < 0) {
    position.y = 0;
    velocity.y *= -1;
  }
  if (position.y > height) {
    position.y = height;
    velocity.y *= -1;
  }
}

function firstForce() {
  let force = createVector(random(-5, 5), random(-5, 5));
  accel.add(force);
}

function mousePressed() {
  firstForce();
}

// Rectangle 클래스

class Rectangle {
  constructor(x, y, w, h) {
    this.x = x;
    this.y = y;
    this.width = w;
    this.height = h;
  }

  display() {
    rect(this.x, this.y, this.width, this.height);
  }

  contains(x, y) {
    if (x > this.x && x < this.x + this.width && y > this.y && y < this.y + this.height) {
      return true;
    } else {
      return false;
    }
  }
}
