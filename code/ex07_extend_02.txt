// 랜덤값이 password(=5)와 같아질 때까지 랜덤값 출력

let password = 0;

function setup() {
  createCanvas(500, 500);
  background(200);

  textSize(30);
  textAlign(CENTER);
  while (password !== 5) {
    password = int(random(-10, 10));
    fill(0);
    text(password, random(width), random(height));
  }
  fill(255, 0, 0);
  text('Welcome!', width / 2, height / 2);
}
