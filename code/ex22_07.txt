let slider1, slider2;

function setup() {
  createCanvas(500, 500);
  slider1 = createSlider(0, 255, 100);
  slider1.position(10, 10);
  slider1.size(200);
  
  slider2 = createSlider(0, 255, 100);
  slider2.position(10, 40);
  slider2.size(200);
}

function draw() { 
  let value1 = slider1.value();
  let c1 = color(value1, 0, 0);
  background(c1);
  
  let value2 = slider2.value();
  let c2 = color(0, value2, 0);
  fill(c2);
  ellipse(width/2, height/2, 300, 300);

  fill(255);
  textSize(20);
  text("Background color", slider1.x+slider1.width+10, 25);
  text("Circle color", slider2.x+slider2.width+10, 55);
}
