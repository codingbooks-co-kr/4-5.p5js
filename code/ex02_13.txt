// 도(degree)를 이용하여 다양한 각의 원호를 그리기1

function setup() {
  createCanvas(500, 200);
  background(200);

  arc(100, 100, 80, 80, radians(0), radians(90));
  arc(200, 100, 80, 80, radians(0), radians(180)); 
  arc(300, 100, 80, 80, radians(0), radians(270));
  arc(400, 100, 80, 80, radians(0), radians(360));
}
