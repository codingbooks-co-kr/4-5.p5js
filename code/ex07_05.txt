// 랜덤한 색상, 위치 및 크기를 가진 사각형 100개 그리기

function setup() {
  createCanvas(500, 500);
  frameRate(2);
}

function draw() {
  background(0);
  for (let i = 0; i < 100; i++) {
    fill(random(255), random(255), random(255), random(255));
    let x = random(width);
    let y = random(height);
    let w = random(width);
    let h = random(height);
    rect(x, y, w, h);
  }
}
