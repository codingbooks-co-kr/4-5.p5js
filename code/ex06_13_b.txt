// 세 개의 문자(a, b, c) 입력 및 출력2

function setup() {
  createCanvas(300, 300);
  textSize(100);
  textAlign(CENTER);
}

function draw() {
  background(0);
  fill(255);
  switch(key) {
  case 'a':
    text('a', width/2, height/2);
    break;
  case 'b':
    text('b', width/2, height/2);
    break;
  case 'c':
    text('c', width/2, height/2);
    break;
  default:
    text('None', width/2, height/2);
    break;
  }
}
