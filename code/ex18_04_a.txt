// 원의 진동 (p5.Vector 클래스 미사용)

let theta = 0.0;

function setup() {
  createCanvas(500, 500);
  background(0);
}

function draw() {
  let x = width/2 + width/2 * cos(theta);
  let y = height/2 + height/2 * sin(theta);
  fill(255, 0, 0);
  ellipse(x, height/2, 30, 30);
  fill(0, 255, 0);
  ellipse(width/2, y, 30, 30);
  theta += 0.05;
}
