// 세 개의 특정 이름만 출력하고 이외는 None 출력

let name = "None";

function setup() {
  createCanvas(400, 200);
  fill(255);
  textSize(50);
  textAlign(CENTER);
}

function draw() {
  background(0);
  switch(name) {
  case "kim":
    text(name, 100, height/2);
    break;
  case "lee":
    text(name, 200, height/2);
    break;
  case "park":
    text(name, 300, height/2);
    break;
  default:
    text("None", width/2, height/2);
    break;
  }
}

function mousePressed() {
  name = prompt("Enter your name");
}
