// mouseMoved() 이벤트 함수 사용

let num = 50; 
let x = [], y = [];

function setup() {
  createCanvas(500, 500);
  for (let i = 0; i < num; i++) {
    x[i] = 0;
    y[i] = 0;
  }
}

function draw() {
  background(0);
  for (let i = 0; i < num; i++) {
    fill(255, 255, 0);
    textSize(20);
    text(i, x[i], y[i]);
  }
}

function mouseMoved() {
  for (let i = 0; i < num-1; i++) {
    x[i] = x[i+1];
    y[i] = y[i+1];
  }
  x[num-1] = mouseX;
  y[num-1] = mouseY;
}
