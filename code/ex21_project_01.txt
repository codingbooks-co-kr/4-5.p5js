// 상하좌우 반사하는 100개의 원 객체들의 크기가 사운드의 크기에 따라 변화

let audio, amp;
let c = [];

function preload() {
  audio = loadSound("hometown.mp3");
}

function setup() {
  createCanvas(500, 500);
  for (let i = 0; i < 100; i++) {
    c[i] = new Circle();
  }
  amp = new p5.Amplitude();
  audio.loop();
}

function draw() {
  // 페이딩 효과 구현
  noStroke();
  fill(0, 10);
  rect(0, 0, width, height);

  let level = amp.getLevel() * 100;
  for (let i = 0; i < c.length; i++) {
    c[i].move();
    c[i].display(level);
    c[i].bounce();
  }
}

class Circle {
  constructor() {
    this.position = createVector(width/2, height/2);
    this.velocity = createVector(random(-1, 1), random(-1, 1));
    this.col = color(random(255), random(255), random(255));
  }

  move() {
    this.position.add(this.velocity);
  }

  display(d) {
    fill(this.col);
    ellipse(this.position.x, this.position.y, d, d);
  }

  bounce() {
    // 경계에 부딪힐 때 반사 적용
    if (this.position.x < 0) {
      this.position.x = 0;
      this.velocity.x *= -1;
    }
    if (this.position.x > width) {
      this.position.x = width;
      this.velocity.x *= -1;
    }
    if (this.position.y < 0) {
      this.position.y = 0;
      this.velocity.y *= -1;
    }
    if (this.position.y > height) {
      this.position.y = height;
      this.velocity.y *= -1;
    }
  }
}
