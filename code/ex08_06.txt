// 마우스를 따라 움직이는 다각형

let points = 4.0, angle = 360.0 / points;

function setup() {
  createCanvas(500, 500);
  background(0);
}

function draw() {
  fill(255, 255, 0);
  beginShape();
  for (let degree = 0; degree < 360; degree += angle) {
    let theta = radians(degree);
    let x = mouseX + 30.0 * cos(theta);
    let y = mouseY + 30.0 * sin(theta);
    vertex(x, y);
  }
  endShape(CLOSE);
}
