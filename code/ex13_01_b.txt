// 원 좌우 반사 (함수 사용)

let c, x, y, d, velocity;

function setup() {
  createCanvas(300, 300);

  c = 255;
  x = width/2;
  y = height/2;
  d = 30;
  velocity = 1;
}

function draw() {
  background(0);
  display();	
  move();
}

function display() {
  fill(c, 255, 0);
  ellipse(x, y, d, d);
}

function move() {
  x += velocity;
  if (x < 0 || x > width) {
    velocity *= -1;
  }
}
