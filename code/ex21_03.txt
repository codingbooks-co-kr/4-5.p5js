// 원 안에서 왼쪽 마우스: 사운드 재생, 오른쪽 마우스: 처음부터 재생, 원 밖: 일시 정지

let audio;
let c, r = 100;

function preload() {
  audio = loadSound("hometown.mp3");
}

function setup() {
  createCanvas(500, 500);
  c = color(0, 0, 255);
}

function draw() {
  background(0);
  fill(c);
  ellipse(width/2, height/2, r*2, r*2);
}

function mousePressed() {
 let d = dist(mouseX, mouseY, width/2, height/2);
  if (d < r && mouseButton === LEFT) {
    c = color(255, 0, 0);
    audio.playMode("untilDone");
    audio.play();
  } else if (d < r && mouseButton === RIGHT) {
    c = color(0, 255, 0);
    audio.playMode("restart");
    audio.play();
  } else {
    c = color(0, 0, 255);
    audio.pause();
  }
}
