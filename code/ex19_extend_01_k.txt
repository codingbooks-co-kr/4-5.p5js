function setup() {
  let array = [10, 20, 30, 40];
  let result = array.splice(2);	// array[2]부터 끝까지 삭제
  print(array);		// 원본 배열은 변경됨
  print(result);		// 삭제된 배열요소 반환

  array = [10, 20, 30, 40];
  array.splice(1, 2); 	  	// array[1]부터 두 개 요소 삭제
  print(array);

  array = [10, 20, 30, 40];
  array.splice(1, 1, 50);  	// array[1] 삭제, 50으로 대체
  print(array);
  
  array = [10, 20, 30, 40];
  array.splice(1, 0, 100); 	// array[1]에 요솟값 100 추가
  print(array);
}
