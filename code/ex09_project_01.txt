// 끊임없이 이동하는 1000개의 점들

let num = 1000;
let x = [], y = [], c = [];

function setup() {
  createCanvas(500, 500);
  for (let i = 0; i < num; i++) {
    x[i] = random(width);
    y[i] = random(height);
    c[i] = color(random(255), random(255), random(255));
  }
}

function draw() {
  // 페이딩 효과
  noStroke();
  fill(0, 10);
  rect(0, 0, width, height);

  for (let i = 0; i < num; i++) {
    stroke(c[i]);
    strokeWeight(5); 
    point(x[i], y[i]);
    x[i] += random(1.0);
    y[i] += random(1.0);

    if (x[i] < 0) {
      x[i] = width;
    }
    if (x[i] > width) {
      x[i] = 0;
    }
    if (y[i] < 0) {
      y[i] = height;
    }
    if (y[i] > height) {
      y[i] = 0;
    }
  }
}
