// 좌우 반사하는 팩맨 만들기
// prompt() 메소드를 사용하여 개폐 각도 수정

let x = 0, degree = 0, maxDegree = 40, velocity = 1; 

function setup() {
  createCanvas(500, 500);
}

function draw() {
  background(0);
  if (velocity > 0) {
    arc(x, height/2, 150, 150, radians(degree), radians(360-degree));
  } 
  if (velocity < 0) {
    arc(x, height/2, 150, 150, radians(180+degree), radians(3*180-degree));
  }

  x += velocity;
  degree++;
  if (x < 0 || x > width) {
    velocity *= -1;
  }
  if (degree >= maxDegree) {
    degree = 0;
  }
}

function mousePressed() {
  let angle = prompt("Enter an angle");
  maxDegree = int(angle);
}
