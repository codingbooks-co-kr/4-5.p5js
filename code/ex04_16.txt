// noise() 함수를 사용하여 선을 랜덤하게 이어서 그리기

let x = 0, y = 0, tx = 0.0, ty = 0.5;

function setup() {
  createCanvas(500, 500);
  background(0);
}

function draw() {
  let px = x;
  let py = y;
  x = noise(tx) * width;
  y = noise(ty) * height;
  stroke(255);
  line(px, py, x, y);
  tx += 0.01;
  ty += 0.01;
}
