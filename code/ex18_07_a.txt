// 상하좌우 반사하는 원 (p5.Vector 클래스 미사용)

let x = 250, y = 250;
let velocity_x = 2, velocity_y = 3;

function setup() {
  createCanvas(500, 500);
  background(0);
}

function draw() {
  ellipse(x, y, 30, 30);
  x += velocity_x;
  y += velocity_y;
  if (x < 0 || x > width) {
    velocity_x *= -1;
  }
  if (y < 0 || y > height) {
    velocity_y *= -1;
  }
}
