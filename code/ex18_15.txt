// 마우스를 누르면 랜덤 이동하다가 감쇄하며 멈추는 원

let position, velocity;
let d = 30, friction = 0.99;

function setup() {
  createCanvas(500, 500); 
  position = createVector(width/2, 50);
  velocity = createVector(random(-5, 5), random(-5, 5));
}

function draw() {
  // 페이딩 효과
  noStroke();
  fill(0, 10);
  rect(0, 0, width, height);

  fill(255);
  ellipse(position.x, position.y, d, d);

  velocity.mult(friction);
  position.add(velocity);

  // 경계에 부딪힐 때 반사 적용
  if (position.x < 0) {
    position.x = 0;
    velocity.x *= -1;
  }
  if (position.x > width) {
    position.x = width;
    velocity.x *= -1;
  }
  if (position.y < 0) {
    position.y = 0;
    velocity.y *= -1;
  }
  if (position.y > height) {
    position.y = height;
    velocity.y *= -1;
  }
}

// 마우스를 누르면 (x, y) 속도를 –5∼5 사이의 값으로 랜덤 설정
function mousePressed() {
  velocity.set(random(-5, 5), random(-5, 5));
}
