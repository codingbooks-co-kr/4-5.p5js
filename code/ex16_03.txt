// 본문 코드 (다형성을 배열에 적용)
// 원, 사각형, 삼각형 객체 좌우 반사 

let sh = [];	

function setup() {
  createCanvas(300, 300);
  sh[0] = new Circle(255, 0, 70, 30, 1);
  sh[1] = new Square(255, 0, 140, 30, 2);
  sh[2] = new Triangle(255, 0, 210, 30, 3);
}

function draw() {
  background(0);
  for (let i = 0; i < sh.length; i++) {
    sh[i].display();
    sh[i].move();
  }
}

// 클래스 코드 

class Shape {
  constructor(c, x, y, d, velocity) {
    this.c = c;
    this.x = x;
    this.y = y;
    this.d = d;
    this.velocity = velocity;
  }

  display() {
    fill(this.c, this.c, 0);
    textSize(this.d);
    text("Shape", this.x, this.y);
  }

  move() {
    this.x += this.velocity;
    if (this.x < 0 || this.x > width) {
      this.velocity *= -1;
    }
  }
}

class Circle extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(this.c, 0, 0);
    ellipse(this.x, this.y, this.d, this.d);
  }
}

class Square extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(0, this.c, 0);
    rect(this.x, this.y, this.d, this.d);
  }
}

class Triangle extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(0, 0, this.c);
    triangle(this.x, this.y, this.x-this.d, this.y+this.d, this.x+this.d, this.y+this.d);
  }
}
