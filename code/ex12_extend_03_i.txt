// ❸ 유형3: 오른쪽 이동하는 숫자 배열

let num = 10;
let x = [], y = [];

function setup() {
  createCanvas(500, 300);
  for (let i = 0; i < num; i++) {
    x[i] = random(-300, 0);
    y[i] = random(0, height);
  }
}

function draw() {
  background(0);
  let xx = move();	// 유형3
  textSize(20);
  text(int(xx), 30, 30);
}

function move() {
  for (let i = 0; i < num; i++) {
    fill(255);
    textSize(30);
    text(i, x[i], y[i]);
    x[i]++;
  }
  return x;
}
