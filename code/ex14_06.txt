// 본문 코드: 원 객체 다섯 개 좌우 반사 (객체 배열)

let c = [];

function setup() {
  createCanvas(300, 300);

  for (let i = 0; i < 5; i++) { 
    c[i] = new Circle(255, 0, i*70, 30, i/2);
  }
}

function draw() {
  background(0);
  for (let i = 0; i < c.length; i++) { 
    c[i].display();
    c[i].move();
  }
}

// Circle 클래스 코드

class Circle {
  constructor(c, x, y, d, velocity) {
    this.c = c;
    this.x = x;
    this.y = y;
    this.d = d;
    this.velocity = velocity;
  }

  display() {
    fill(this.c, 255, 0);
    ellipse(this.x, this.y, this.d, this.d);
  }

  move() {
    this.x += this.velocity;
    if (this.x < 0 || this.x > width) {
      this.velocity *= -1;
    }
  }
}
