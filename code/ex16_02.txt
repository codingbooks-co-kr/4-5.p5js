// 본문 코드: 문자열, 원, 사각형, 삼각형 객체의 좌우 반사

let sh0, sh1, sh2, sh3;

function setup() {
  createCanvas(300, 300);
  sh0 = new Shape(255, 0, 30, 30, 1);
  sh1 = new Circle(255, 0, 70, 30, 2);
  sh2 = new Square(255, 0, 140, 30, 3);
  sh3 = new Triangle(255, 0, 210, 30, 4);
}

function draw() {
  background(0);
  sh0.display();
  sh0.move();
  sh1.display();
  sh1.move();
  sh2.display();
  sh2.move();
  sh3.display();
  sh3.move();
}

// 클래스 코드

class Shape {
  constructor(c, x, y, d, velocity) {
    this.c = c;
    this.x = x;
    this.y = y;
    this.d = d;
    this.velocity = velocity;
  }

  display() {
    fill(this.c, this.c, 0);
    textSize(this.d);
    text("Shape", this.x, this.y);
  }

  move() {
    this.x += this.velocity;
    if (this.x < 0 || this.x > width) {
      this.velocity *= -1;
    }
  }
}

class Circle extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(this.c, 0, 0);
    ellipse(this.x, this.y, this.d, this.d);
  }
}

class Square extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(0, this.c, 0);
    rect(this.x, this.y, this.d, this.d);
  }
}

class Triangle extends Shape {
  constructor(c, x, y, d, velocity) {
    super(c, x, y, d, velocity);
  }

  display() {
    fill(0, 0, this.c);
    triangle(this.x, this.y, this.x-this.d, this.y+this.d, this.x+this.d, this.y+this.d);
  }
}
