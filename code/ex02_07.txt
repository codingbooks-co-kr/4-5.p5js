// 문자열을 왼쪽, 오른쪽 및 가운데 정렬하여 출력 화면에 표시

function setup() {
  createCanvas(300, 300);
  background(200);

  line(150, 0, 150, 300);
  textSize(30);
  line(0, 100, 300, 100);
  textAlign(LEFT);
  text("Hello", 150, 100);

  line(0, 150, 300, 150);
  textAlign(RIGHT);
  text("Hello", 150, 150);

  line(0, 200, 300, 200);
  textAlign(CENTER);
  text("Hello", 150, 200);

  line(0, 250, 300, 250);
  textAlign(CENTER, CENTER);
  text("Hello", 150, 250);
}
