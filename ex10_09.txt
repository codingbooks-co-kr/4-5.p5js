// 2차원 색상 배열을 사용한 원의 색상 변화

let c = [];

function setup() {
  createCanvas(300, 300);
  for (let y = 0; y < 5; y++) {
    c[y] = [];
    for (let x = 0; x < 5; x++) {
      c[y][x] = 16 * y * x;
      c[y][x] = constrain(c[y][x], 0, 255);
    }
  }
}

function draw() {
  background(0);
  for (let y = 0; y < 5; y++) {
    for (let x = 0; x < 5; x++) {
      fill(c[y][x], 0, 255);
      ellipseMode(CORNER);
      ellipse(x * 60, y * 60, 60, 60);
    }
  }
}
