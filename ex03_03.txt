// 다양한 크기의 원 그리기

let x = 100, y = 100, d = 80;

function setup() {
  createCanvas(300, 300);
  background(200);

  noFill();
  ellipse(x%30, y%30, d%30, d%30);
  ellipse(x/2, y/2, d/2, d/2);
  ellipse(x-30, y-30, d-30, d-30);
  ellipse(x, y, d, d);
  ellipse(x+30, y+30, d+30, d+30);
  ellipse(x*2, y*2, d*2, d*2);
}
