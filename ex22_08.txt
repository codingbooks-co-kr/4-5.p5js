// 입력 다이얼로그창에 특정 이름이 입력되는 경우에만 해당 이름 출력, 이외는 None 출력

let inp, name = "None";

function setup() {
  createCanvas(500, 500);
  inp = createInput('');
  inp.position(10, 40);
  inp.size(100);
}

function draw() {
  background(0);
  fill(255);
  textSize(20);
  text("Input name!", 10, 30);

  let name = inp.value();
  textSize(50);
  switch(name) {
  case "kim":
    text(name, 100, 300);
    break;
  case "lee":
    text(name, 200, 300);
    break;
  case "park":
    text(name, 300, 300);
    break;
  default:
    text("None", 250, 300);
    break;
  }
}
