// <if-if-if>문

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  if (mouseY < height/3) {
    rect(0, 0, width, height/3);
  }
  if (mouseY > height/3 && mouseY < height*2/3) {
    rect(0, height/3, width, height/3);
  }
  if (mouseY >= height*2/3) {
    rect(0, height*2/3, width, height/3);
  }
}
