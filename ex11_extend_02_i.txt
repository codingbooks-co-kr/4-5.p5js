// 매개변수가 없는 콜백함수

let r = 100.0;

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  // ❶distance는 콜백함수
  show(distance);
}

// ❷매개변수 d는 콜백함수를 전달받음
function show(d) {
  fill(255);
  textSize(20);
  text("콜백함수", 10, 30);

  // ❸콜백함수 호출
  d();
}

// 콜백함수로 익명함수(유형1)를 사용
let distance = function () {
  let x = width / 2;
  let y = height / 2;
  let d = dist(x, y, mouseX, mouseY);
  fill(d, d, 255);
  ellipse(x, y, r*2, r*2);
  print(d);
};
